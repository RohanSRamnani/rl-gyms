# domains/minirl/physics.py
"""Pure physics functions for MiniRL. No RLGym types appear here."""
import numpy as np

from domains.minirl.state import (
    BOOST_ACCEL, BOOST_CONSUMPTION, BRAKE_ACCEL, CAR_MAX_SPEED, CAR_RADIUS,
    COAST_DECEL, MAX_TURN_RATE, SIDE_WALL_X, BACK_WALL_Y, THROTTLE_ACCEL,
    Car, heading, wrap_angle,
)


def step_car(car: Car, throttle: float, steer: float, boost_input: float, dt: float) -> None:
    """Advance one car by dt seconds. Mutates `car` in place.

    The car is an arcade car: velocity always points along its heading.
    """
    fwd = heading(car.yaw)
    forward_speed = float(np.dot(car.vel, fwd))

    # Turning degrades with speed — the one handling behaviour worth preserving.
    turn_scale = 1.0 - 0.5 * min(abs(forward_speed) / CAR_MAX_SPEED, 1.0)
    car.yaw = wrap_angle(car.yaw + steer * MAX_TURN_RATE * turn_scale * dt)
    fwd = heading(car.yaw)

    boosting = boost_input > 0.5 and car.boost > 0.0
    if boosting:
        car.boost = max(0.0, car.boost - BOOST_CONSUMPTION * dt)
        accel = THROTTLE_ACCEL + BOOST_ACCEL
    elif throttle == 0.0:
        # Coast toward zero, but never overshoot into reverse.
        decel = COAST_DECEL * dt
        accel = (-forward_speed / dt) if abs(forward_speed) < decel \
            else -np.sign(forward_speed) * COAST_DECEL
    elif forward_speed * throttle < 0.0:
        accel = throttle * BRAKE_ACCEL
    else:
        accel = throttle * THROTTLE_ACCEL

    forward_speed = float(np.clip(forward_speed + accel * dt, -CAR_MAX_SPEED, CAR_MAX_SPEED))
    car.vel = fwd * forward_speed
    car.pos = car.pos + car.vel * dt
    _clamp_car_to_arena(car)


def _clamp_car_to_arena(car: Car) -> None:
    limit_x = SIDE_WALL_X - CAR_RADIUS
    limit_y = BACK_WALL_Y - CAR_RADIUS
    for axis, limit in ((0, limit_x), (1, limit_y)):
        if abs(car.pos[axis]) > limit:
            car.pos[axis] = np.sign(car.pos[axis]) * limit
            car.vel[axis] = 0.0
```

def _clamp_speed(vel: np.ndarray, max_speed: float) -> np.ndarray:
    speed = float(np.linalg.norm(vel))
    if speed > max_speed:
        return vel * (max_speed / speed)
    return vel


def step_ball(state: "MiniRLState", dt: float) -> None:
    """Advance the ball by dt seconds, resolving walls and goals."""
    state.ball_vel = state.ball_vel * ((1.0 - BALL_DRAG) ** dt)
    state.ball_vel = _clamp_speed(state.ball_vel, BALL_MAX_SPEED)
    state.ball_pos = state.ball_pos + state.ball_vel * dt

    # Side walls always bounce.
    limit_x = SIDE_WALL_X - BALL_RADIUS
    if abs(state.ball_pos[0]) > limit_x:
        state.ball_pos[0] = np.sign(state.ball_pos[0]) * limit_x
        state.ball_vel[0] *= -WALL_RESTITUTION

    # Back walls bounce unless the ball is inside the goal mouth.
    limit_y = BACK_WALL_Y - BALL_RADIUS
    if abs(state.ball_pos[1]) > limit_y:
        if abs(state.ball_pos[0]) < GOAL_HALF_WIDTH:
            if state.ball_pos[1] >= BACK_WALL_Y - BALL_RADIUS:
                state.scoring_team = 0     # blue attacks +y
            else:
                state.scoring_team = 1     # orange attacks -y
        else:
            state.ball_pos[1] = np.sign(state.ball_pos[1]) * limit_y
            state.ball_vel[1] *= -WALL_RESTITUTION
```

```python
def collide_car_ball(state: "MiniRLState", agent_id: int, tick: int) -> bool:
    """Resolve a circle-circle car/ball contact. Returns True if contact occurred."""
    car = state.cars[agent_id]
    delta = state.ball_pos - car.pos
    dist = float(np.linalg.norm(delta))
    min_dist = CAR_RADIUS + BALL_RADIUS

    if dist >= min_dist:
        return False
    if dist < 1e-9:
        # Degenerate exact overlap — push out along the car's heading.
        normal = heading(car.yaw)
    else:
        normal = delta / dist

    # Separate first, so the impulse is applied from a clean, non-overlapping state.
    state.ball_pos = car.pos + normal * min_dist

    # Reflect the closing component of relative velocity, then add a constant
    # hit force so that even a stationary car nudges the ball.
    relative = state.ball_vel - car.vel
    closing = float(np.dot(relative, normal))
    if closing < 0.0:
        state.ball_vel = state.ball_vel - (1.0 + WALL_RESTITUTION) * closing * normal
    state.ball_vel = state.ball_vel + normal * BALL_HIT_FORCE
    state.ball_vel = _clamp_speed(state.ball_vel, BALL_MAX_SPEED)

    state.last_touch_tick = tick
    return True
```

# domains/minirl/engine.py
"""The MiniRL transition engine — the only module that advances physics."""
from typing import Any, Dict, List

import numpy as np
from rlgym.api import TransitionEngine

from domains.minirl.physics import collide_car_ball, step_ball, step_car
from domains.minirl.state import DT, MiniRLState


class MiniRLEngine(TransitionEngine[int, MiniRLState, np.ndarray]):
    """Advances MiniRL physics.

    An engine action for one agent has shape (n_ticks, 3): one
    [throttle, steer, boost] row per physics tick. `step` walks those rows,
    which is how RepeatAction's tiling reaches the simulator.
    """

    def __init__(self) -> None:
        self._state: MiniRLState = self.create_base_state()
        self._config: Dict[str, Any] = {}

    @property
    def agents(self) -> List[int]:
        return list(self._state.cars.keys())

    @property
    def max_num_agents(self) -> int:
        return 2

    @property
    def state(self) -> MiniRLState:
        return self._state

    @property
    def config(self) -> Dict[str, Any]:
        return self._config

    @config.setter
    def config(self, value: Dict[str, Any]) -> None:
        self._config = value

    def create_base_state(self) -> MiniRLState:
        return MiniRLState(ball_pos=np.zeros(2), ball_vel=np.zeros(2), cars={})

    def set_state(self, desired_state: MiniRLState, shared_info: Dict[str, Any]) -> MiniRLState:
        self._state = desired_state
        return self._state

    def step(self, actions: Dict[int, np.ndarray], shared_info: Dict[str, Any]) -> MiniRLState:
        n_ticks = len(next(iter(actions.values())))
        for tick_index in range(n_ticks):
            for agent_id, car in self._state.cars.items():
                throttle, steer, boost = actions[agent_id][tick_index]
                step_car(car, float(throttle), float(steer), float(boost), DT)

            step_ball(self._state, DT)
            for agent_id in self._state.cars:
                collide_car_ball(self._state, agent_id, self._state.ticks)

            self._state.ticks += 1
            if self._state.scoring_team is not None:
                break
        return self._state

    def close(self) -> None:
        pass

# domains/minirl/mutators.py
"""State mutators — these write the initial state, and are where curriculum lives."""
from typing import Any, Dict, Optional

import numpy as np
from rlgym.api import StateMutator

from domains.minirl.state import BACK_WALL_Y, SIDE_WALL_X, Car, MiniRLState


class FixedTeamSizeMutator(StateMutator[MiniRLState]):
    """Populates state.cars with the requested number of cars per team."""

    def __init__(self, blue_size: int = 1, orange_size: int = 0) -> None:
        self.blue_size = blue_size
        self.orange_size = orange_size

    def apply(self, state: MiniRLState, shared_info: Dict[str, Any]) -> None:
        state.cars = {}
        agent_id = 0
        for is_orange in (False, True):
            count = self.orange_size if is_orange else self.blue_size
            for _ in range(count):
                state.cars[agent_id] = Car(pos=np.zeros(2), vel=np.zeros(2), yaw=0.0,
                                           boost=100.0, is_orange=is_orange)
                agent_id += 1


class KickoffMutator(StateMutator[MiniRLState]):
    """Centres the ball and scatters each car randomly within its own half.

    Randomized spawns stop the agent memorizing a single trajectory, and this is
    the hook where a curriculum would later live.
    """

    def __init__(self, rng: Optional[np.random.Generator] = None) -> None:
        self.rng = rng if rng is not None else np.random.default_rng()

    def apply(self, state: MiniRLState, shared_info: Dict[str, Any]) -> None:
        state.ball_pos = np.zeros(2)
        state.ball_vel = np.zeros(2)
        state.ticks = 0
        state.last_touch_tick = None
        state.scoring_team = None

        for car in state.cars.values():
            side = -1.0 if not car.is_orange else 1.0
            x = self.rng.uniform(-0.75 * SIDE_WALL_X, 0.75 * SIDE_WALL_X)
            y = side * self.rng.uniform(0.2 * BACK_WALL_Y, 0.8 * BACK_WALL_Y)
            car.pos = np.array([x, y], dtype=float)
            car.vel = np.zeros(2)
            car.yaw = float(self.rng.uniform(-np.pi, np.pi))
            car.boost = 100.0


class MutatorSequence(StateMutator[MiniRLState]):
    """Applies mutators in order."""

    def __init__(self, *mutators: StateMutator[MiniRLState]) -> None:
        self.mutators = mutators

    def apply(self, state: MiniRLState, shared_info: Dict[str, Any]) -> None:
        for mutator in self.mutators:
            mutator.apply(state, shared_info)
```

# domains/minirl/obs.py
"""Observation builder — turns state into what each agent actually sees."""
from typing import Any, Dict, List

import numpy as np
from rlgym.api import ObsBuilder

from domains.minirl.state import (
    BALL_MAX_SPEED, CAR_MAX_SPEED, POS_NORM, MiniRLState, wrap_angle,
)

OBS_SIZE = 16


class DefaultObs(ObsBuilder[int, np.ndarray, MiniRLState, int]):
    """A 16-dimensional normalized observation.

    Layout:
        0:2   car position          8:10  ball position
        2:4   car velocity         10:12  ball velocity
        4:6   car heading (cos,sin) 12:14 ball position relative to car
        6:7   car boost            14:16 ball velocity relative to car, then distance

    Yaw is encoded as (cos, sin) rather than a raw angle: a raw angle has a
    discontinuity at +/-pi that quietly wrecks learning.
    """

    def get_obs_space(self, agent: int) -> int:
        return OBS_SIZE

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def build_obs(self, agents: List[int], state: MiniRLState,
                  shared_info: Dict[str, Any]) -> Dict[int, np.ndarray]:
        return {agent: self._build_one(agent, state) for agent in agents}

    def _build_one(self, agent: int, state: MiniRLState) -> np.ndarray:
        car = state.cars[agent]
        # Orange sees a point-mirrored world, so both teams face the same problem.
        mirror = -1.0 if car.is_orange else 1.0

        car_pos = car.pos * mirror
        car_vel = car.vel * mirror
        ball_pos = state.ball_pos * mirror
        ball_vel = state.ball_vel * mirror
        yaw = wrap_angle(car.yaw + np.pi) if car.is_orange else car.yaw

        rel_pos = ball_pos - car_pos
        rel_vel = ball_vel - car_vel
        distance = float(np.linalg.norm(rel_pos))

        obs = np.concatenate([
            car_pos / POS_NORM,
            car_vel / CAR_MAX_SPEED,
            [np.cos(yaw), np.sin(yaw)],
            [car.boost / 100.0],
            ball_pos / POS_NORM,
            ball_vel / BALL_MAX_SPEED,
            rel_pos / POS_NORM,
            rel_vel / CAR_MAX_SPEED,
            [distance / 10000.0],
        ])
        return obs.astype(np.float32)
```

# domains/minirl/actions.py
"""Action parsers — these define the vocabulary the policy speaks in."""
import itertools
from typing import Any, Dict, List

import numpy as np
from rlgym.api import ActionParser

from domains.minirl.state import ACTION_REPEAT, MiniRLState

THROTTLE_OPTIONS = (-1.0, 0.0, 1.0)
STEER_OPTIONS = (-1.0, -0.5, 0.0, 0.5, 1.0)
BOOST_OPTIONS = (0.0, 1.0)
N_ACTIONS = len(THROTTLE_OPTIONS) * len(STEER_OPTIONS) * len(BOOST_OPTIONS)  # 30


class LookupTableAction(ActionParser[int, int, np.ndarray, MiniRLState, int]):
    """Maps a discrete action index onto a [throttle, steer, boost] triple.

    Discretizing keeps the policy a clean categorical distribution, which is
    both easier to learn and easier to reason about than continuous control.
    """

    def __init__(self) -> None:
        self._table = np.array(
            list(itertools.product(THROTTLE_OPTIONS, STEER_OPTIONS, BOOST_OPTIONS)),
            dtype=np.float32,
        )

    def get_action_space(self, agent: int) -> int:
        return N_ACTIONS

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def parse_actions(self, actions: Dict[int, int], state: MiniRLState,
                      shared_info: Dict[str, Any]) -> Dict[int, np.ndarray]:
        return {agent: self._table[int(action)] for agent, action in actions.items()}


class RepeatAction(ActionParser[int, int, np.ndarray, MiniRLState, int]):
    """Holds one decision for `repeats` physics ticks.

    This shortens the effective horizon by `repeats` and is one of the highest
    leverage knobs in the whole configuration.
    """

    def __init__(self, parser: ActionParser, repeats: int = ACTION_REPEAT) -> None:
        self.parser = parser
        self.repeats = repeats

    def get_action_space(self, agent: int) -> int:
        return self.parser.get_action_space(agent)

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        self.parser.reset(agents, initial_state, shared_info)

    def parse_actions(self, actions: Dict[int, int], state: MiniRLState,
                      shared_info: Dict[str, Any]) -> Dict[int, np.ndarray]:
        parsed = self.parser.parse_actions(actions, state, shared_info)
        return {agent: np.tile(control, (self.repeats, 1))
                for agent, control in parsed.items()}
```

# domains/minirl/rewards.py
"""Reward functions.

The weights used in env.py form a deliberate ladder spanning three orders of
magnitude. Dense shaping terms are training wheels: they must stay small enough
that farming them can never beat actually scoring.
"""
from typing import Any, Dict, List, Tuple

import numpy as np
from rlgym.api import RewardFunction

from domains.minirl.state import BACK_WALL_Y, BALL_MAX_SPEED, CAR_MAX_SPEED, MiniRLState


class SpeedTowardBallReward(RewardFunction[int, MiniRLState, float]):
    """Rewards closing speed toward the ball, in [0, 1]."""

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def get_rewards(self, agents: List[int], state: MiniRLState,
                    is_terminated: Dict[int, bool], is_truncated: Dict[int, bool],
                    shared_info: Dict[str, Any]) -> Dict[int, float]:
        rewards = {}
        for agent in agents:
            car = state.cars[agent]
            offset = state.ball_pos - car.pos
            distance = float(np.linalg.norm(offset))
            if distance < 1e-6:
                rewards[agent] = 0.0
                continue
            closing = float(np.dot(car.vel, offset / distance))
            rewards[agent] = max(closing / CAR_MAX_SPEED, 0.0)
        return rewards


class VelocityBallToGoalReward(RewardFunction[int, MiniRLState, float]):
    """Rewards ball velocity toward the opponent goal, in [0, 1]."""

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def get_rewards(self, agents: List[int], state: MiniRLState,
                    is_terminated: Dict[int, bool], is_truncated: Dict[int, bool],
                    shared_info: Dict[str, Any]) -> Dict[int, float]:
        rewards = {}
        for agent in agents:
            car = state.cars[agent]
            goal_y = -BACK_WALL_Y if car.is_orange else BACK_WALL_Y
            offset = np.array([0.0, goal_y]) - state.ball_pos
            distance = float(np.linalg.norm(offset))
            if distance < 1e-6:
                rewards[agent] = 0.0
                continue
            toward = float(np.dot(state.ball_vel, offset / distance))
            rewards[agent] = max(toward / BALL_MAX_SPEED, 0.0)
        return rewards


class TouchReward(RewardFunction[int, MiniRLState, float]):
    """Fires exactly once per ball touch."""

    def __init__(self) -> None:
        self._seen_touch_tick = None

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        self._seen_touch_tick = initial_state.last_touch_tick

    def get_rewards(self, agents: List[int], state: MiniRLState,
                    is_terminated: Dict[int, bool], is_truncated: Dict[int, bool],
                    shared_info: Dict[str, Any]) -> Dict[int, float]:
        touched = (state.last_touch_tick is not None
                   and state.last_touch_tick != self._seen_touch_tick)
        self._seen_touch_tick = state.last_touch_tick
        return {agent: 1.0 if touched else 0.0 for agent in agents}


class GoalReward(RewardFunction[int, MiniRLState, float]):
    """+1 for the scoring team, -1 for the conceding team, 0 otherwise."""

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def get_rewards(self, agents: List[int], state: MiniRLState,
                    is_terminated: Dict[int, bool], is_truncated: Dict[int, bool],
                    shared_info: Dict[str, Any]) -> Dict[int, float]:
        rewards = {}
        for agent in agents:
            if state.scoring_team is None:
                rewards[agent] = 0.0
            else:
                rewards[agent] = 1.0 if state.scoring_team == state.cars[agent].team else -1.0
        return rewards


class CombinedReward(RewardFunction[int, MiniRLState, float]):
    """Weighted sum of component rewards."""

    def __init__(self, *weighted_rewards: Tuple[RewardFunction, float]) -> None:
        self.weighted_rewards = weighted_rewards

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        for reward_fn, _ in self.weighted_rewards:
            reward_fn.reset(agents, initial_state, shared_info)

    def get_rewards(self, agents: List[int], state: MiniRLState,
                    is_terminated: Dict[int, bool], is_truncated: Dict[int, bool],
                    shared_info: Dict[str, Any]) -> Dict[int, float]:
        totals = {agent: 0.0 for agent in agents}
        for reward_fn, weight in self.weighted_rewards:
            component = reward_fn.get_rewards(agents, state, is_terminated, is_truncated,
                                              shared_info)
            for agent in agents:
                totals[agent] += weight * component[agent]
        return totals
```

# domains/minirl/dones.py
"""Done conditions.

GoalCondition TERMINATES: the episode genuinely ended, so V(s') is 0.
The timeouts TRUNCATE: we cut the episode off, so V(s') must be bootstrapped.
Conflating the two silently corrupts value learning.
"""
from typing import Any, Dict, List

from rlgym.api import DoneCondition

from domains.minirl.state import TICK_RATE, MiniRLState


class GoalCondition(DoneCondition[int, MiniRLState]):
    """True once a goal has been scored. This is a TERMINATION."""

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def is_done(self, agents: List[int], state: MiniRLState,
                shared_info: Dict[str, Any]) -> Dict[int, bool]:
        scored = state.scoring_team is not None
        return {agent: scored for agent in agents}


class TimeoutCondition(DoneCondition[int, MiniRLState]):
    """True once the episode has run too long. This is a TRUNCATION."""

    def __init__(self, timeout_seconds: float) -> None:
        self.timeout_ticks = int(timeout_seconds * TICK_RATE)

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def is_done(self, agents: List[int], state: MiniRLState,
                shared_info: Dict[str, Any]) -> Dict[int, bool]:
        expired = state.ticks >= self.timeout_ticks
        return {agent: expired for agent in agents}


class NoTouchTimeoutCondition(DoneCondition[int, MiniRLState]):
    """True once nobody has touched the ball for too long. This is a TRUNCATION."""

    def __init__(self, timeout_seconds: float) -> None:
        self.timeout_ticks = int(timeout_seconds * TICK_RATE)

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        pass

    def is_done(self, agents: List[int], state: MiniRLState,
                shared_info: Dict[str, Any]) -> Dict[int, bool]:
        last_touch = state.last_touch_tick if state.last_touch_tick is not None else 0
        expired = (state.ticks - last_touch) >= self.timeout_ticks
        return {agent: expired for agent in agents}


class AnyCondition(DoneCondition[int, MiniRLState]):
    """True when any child condition is true."""

    def __init__(self, *conditions: DoneCondition) -> None:
        self.conditions = conditions

    def reset(self, agents: List[int], initial_state: MiniRLState,
              shared_info: Dict[str, Any]) -> None:
        for condition in self.conditions:
            condition.reset(agents, initial_state, shared_info)

    def is_done(self, agents: List[int], state: MiniRLState,
                shared_info: Dict[str, Any]) -> Dict[int, bool]:
        result = {agent: False for agent in agents}
        for condition in self.conditions:
            child = condition.is_done(agents, state, shared_info)
            for agent in agents:
                result[agent] = result[agent] or child[agent]
        return result
```

"""Assembles the six configuration objects into an RLGym environment.

Read alongside the official build_rlgym_v2_env() from the Rocket League docs —
this is deliberately the same shape.
"""
from typing import Optional

import numpy as np
from rlgym.api import RLGym

from domains.minirl.actions import N_ACTIONS, LookupTableAction, RepeatAction
from domains.minirl.dones import (
    AnyCondition, GoalCondition, NoTouchTimeoutCondition, TimeoutCondition,
)
from domains.minirl.engine import MiniRLEngine
from domains.minirl.mutators import FixedTeamSizeMutator, KickoffMutator, MutatorSequence
from domains.minirl.obs import OBS_SIZE, DefaultObs
from domains.minirl.rewards import (
    CombinedReward, GoalReward, SpeedTowardBallReward, TouchReward,
    VelocityBallToGoalReward,
)
from domains.minirl.state import ACTION_REPEAT

__all__ = ["build_env", "OBS_SIZE", "N_ACTIONS"]

NO_TOUCH_TIMEOUT_SECONDS = 10.0
GAME_TIMEOUT_SECONDS = 30.0


def build_env(blue_size: int = 1, orange_size: int = 0,
              rng: Optional[np.random.Generator] = None) -> RLGym:
    return RLGym(
        state_mutator=MutatorSequence(
            FixedTeamSizeMutator(blue_size=blue_size, orange_size=orange_size),
            KickoffMutator(rng=rng),
        ),
        obs_builder=DefaultObs(),
        action_parser=RepeatAction(LookupTableAction(), repeats=ACTION_REPEAT),
        reward_fn=CombinedReward(
            (SpeedTowardBallReward(), 0.01),
            (VelocityBallToGoalReward(), 0.1),
            (TouchReward(), 0.5),
            (GoalReward(), 10.0),
        ),
        transition_engine=MiniRLEngine(),
        termination_cond=GoalCondition(),
        truncation_cond=AnyCondition(
            NoTouchTimeoutCondition(timeout_seconds=NO_TOUCH_TIMEOUT_SECONDS),
            TimeoutCondition(timeout_seconds=GAME_TIMEOUT_SECONDS),
        ),
    )
```

# core/ppo.py
"""PPO. Domain-agnostic: sizes are arguments, so this file never imports a domain."""
from dataclasses import dataclass
from typing import Dict, Tuple

import numpy as np
import torch
import torch.nn as nn


@dataclass
class PPOConfig:
    gamma: float = 0.99
    lam: float = 0.95
    clip: float = 0.2
    entropy_coef: float = 0.01
    value_coef: float = 0.5
    epochs: int = 4
    minibatch_size: int = 256
    max_grad_norm: float = 0.5


def _mlp(in_size: int, out_size: int, hidden: int) -> nn.Sequential:
    return nn.Sequential(
        nn.Linear(in_size, hidden), nn.Tanh(),
        nn.Linear(hidden, hidden), nn.Tanh(),
        nn.Linear(hidden, out_size),
    )


class ActorCritic(nn.Module):
    """Separate policy and value networks over a discrete action space."""

    def __init__(self, obs_size: int, n_actions: int, hidden: int = 256) -> None:
        super().__init__()
        self.policy = _mlp(obs_size, n_actions, hidden)
        self.value = _mlp(obs_size, 1, hidden)

    @torch.no_grad()
    def act(self, obs: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        distribution = torch.distributions.Categorical(logits=self.policy(obs))
        actions = distribution.sample()
        return actions, distribution.log_prob(actions), self.value(obs).squeeze(-1)

    @torch.no_grad()
    def predict_values(self, obs: torch.Tensor) -> torch.Tensor:
        return self.value(obs).squeeze(-1)

    def evaluate(self, obs: torch.Tensor, actions: torch.Tensor
                 ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        distribution = torch.distributions.Categorical(logits=self.policy(obs))
        return (distribution.log_prob(actions), distribution.entropy(),
                self.value(obs).squeeze(-1))


def compute_gae(rewards: np.ndarray, values: np.ndarray, bootstraps: np.ndarray,
                dones: np.ndarray, gamma: float, lam: float
                ) -> Tuple[np.ndarray, np.ndarray]:
    """Generalized Advantage Estimation over arrays shaped (T, N).

    `bootstraps[t]` is the value of whatever state actually followed step t:
    0.0 when the episode TERMINATED, V(final_obs) when it was TRUNCATED, and
    V(obs[t+1]) otherwise. Passing it explicitly is what keeps truncation
    correct — reading values[t+1] would use the next episode's first state.
    """
    horizon = rewards.shape[0]
    advantages = np.zeros_like(rewards)
    running = np.zeros(rewards.shape[1], dtype=rewards.dtype)

    for t in reversed(range(horizon)):
        delta = rewards[t] + gamma * bootstraps[t] - values[t]
        running = delta + gamma * lam * (1.0 - dones[t]) * running
        advantages[t] = running

    return advantages, advantages + values


def ppo_update(model: ActorCritic, optimizer: torch.optim.Optimizer,
               batch: Dict[str, torch.Tensor], config: PPOConfig) -> Dict[str, float]:
    """One PPO update over a flat batch. Returns diagnostics."""
    n = batch["obs"].shape[0]
    advantages = batch["advantages"]
    advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)

    stats = {"policy_loss": 0.0, "value_loss": 0.0, "entropy": 0.0}
    updates = 0

    for _ in range(config.epochs):
        for start in range(0, n, config.minibatch_size):
            index = slice(start, start + config.minibatch_size)
            log_probs, entropy, values = model.evaluate(batch["obs"][index],
                                                        batch["actions"][index])

            ratio = torch.exp(log_probs - batch["old_log_probs"][index])
            unclipped = ratio * advantages[index]
            clipped = torch.clamp(ratio, 1.0 - config.clip, 1.0 + config.clip) * advantages[index]
            policy_loss = -torch.min(unclipped, clipped).mean()

            value_loss = ((values - batch["returns"][index]) ** 2).mean()
            entropy_mean = entropy.mean()
            loss = (policy_loss
                    + config.value_coef * value_loss
                    - config.entropy_coef * entropy_mean)

            optimizer.zero_grad()
            loss.backward()
            nn.utils.clip_grad_norm_(model.parameters(), config.max_grad_norm)
            optimizer.step()

            stats["policy_loss"] += float(policy_loss)
            stats["value_loss"] += float(value_loss)
            stats["entropy"] += float(entropy_mean)
            updates += 1

    return {key: value / max(updates, 1) for key, value in stats.items()}
```

# core/metrics.py
"""Per-episode metric tracking. Domain-agnostic."""
from collections import deque
from typing import Dict


class EpisodeTracker:
    """Accumulates per-step signals into per-episode statistics."""

    def __init__(self, window: int = 100) -> None:
        self.returns = deque(maxlen=window)
        self.touches = deque(maxlen=window)
        self.goals = deque(maxlen=window)
        self._current_return = 0.0
        self._current_touched = False

    def record(self, reward: float, terminated: bool, truncated: bool,
               touched: bool, scored: bool) -> None:
        self._current_return += reward
        self._current_touched = self._current_touched or touched
        if terminated or truncated:
            self.returns.append(self._current_return)
            self.touches.append(1.0 if self._current_touched else 0.0)
            self.goals.append(1.0 if scored else 0.0)
            self._current_return = 0.0
            self._current_touched = False

    def snapshot(self) -> Dict[str, float]:
        if not self.returns:
            return {"mean_episode_reward": 0.0, "touch_rate": 0.0,
                    "goal_rate": 0.0, "episodes": 0}
        return {
            "mean_episode_reward": sum(self.returns) / len(self.returns),
            "touch_rate": sum(self.touches) / len(self.touches),
            "goal_rate": sum(self.goals) / len(self.goals),
            "episodes": len(self.returns),
        }

# core/rollout.py
"""Vectorized rollout collection. Domain-agnostic: takes an env factory."""
from typing import Any, Callable, Dict, List, Optional, Tuple

import numpy as np
import torch

from core.metrics import EpisodeTracker
from core.ppo import ActorCritic


class VecEnv:
    """Runs N single-agent RLGym environments in-process with auto-reset.

    On an episode boundary `step` returns the observation of the FINAL state
    before resetting, in `final_obs`. PPO needs that observation to bootstrap
    truncated episodes correctly.
    """

    def __init__(self, env_fn: Callable[[], Any], n_envs: int, agent_id: int = 0) -> None:
        self.envs = [env_fn() for _ in range(n_envs)]
        self.n_envs = n_envs
        self.agent_id = agent_id
        self.last_obs: Optional[np.ndarray] = None   # carried across rollouts

    def reset(self) -> np.ndarray:
        return np.stack([env.reset()[self.agent_id] for env in self.envs])

    def step(self, actions: np.ndarray) -> Tuple[
        np.ndarray, np.ndarray, np.ndarray, np.ndarray, List[Optional[np.ndarray]]
    ]:
        observations, rewards = [], []
        terminated_flags, truncated_flags = [], []
        final_observations: List[Optional[np.ndarray]] = []

        for index, env in enumerate(self.envs):
            obs, reward, terminated, truncated = env.step(
                {self.agent_id: int(actions[index])}
            )
            done_terminated = bool(terminated[self.agent_id])
            done_truncated = bool(truncated[self.agent_id])

            rewards.append(float(reward[self.agent_id]))
            terminated_flags.append(done_terminated)
            truncated_flags.append(done_truncated)

            if done_terminated or done_truncated:
                final_observations.append(obs[self.agent_id])
                observations.append(env.reset()[self.agent_id])
            else:
                final_observations.append(None)
                observations.append(obs[self.agent_id])

        return (np.stack(observations), np.array(rewards, dtype=np.float32),
                np.array(terminated_flags), np.array(truncated_flags),
                final_observations)

    def scoring_teams(self) -> List[Optional[int]]:
        return [env.state.scoring_team for env in self.envs]

    def touched_flags(self) -> List[bool]:
        return [env.state.last_touch_tick is not None for env in self.envs]


def collect_rollout(vec_env: VecEnv, model: ActorCritic, horizon: int,
                    tracker: EpisodeTracker) -> Dict[str, np.ndarray]:
    """Collect `horizon` steps from every env, with correct bootstrap values."""
    n = vec_env.n_envs
    obs_size = model.policy[0].in_features

    obs_buffer = np.zeros((horizon, n, obs_size), dtype=np.float32)
    actions_buffer = np.zeros((horizon, n), dtype=np.int64)
    log_probs_buffer = np.zeros((horizon, n), dtype=np.float32)
    values_buffer = np.zeros((horizon, n), dtype=np.float32)
    rewards_buffer = np.zeros((horizon, n), dtype=np.float32)
    bootstraps_buffer = np.zeros((horizon, n), dtype=np.float32)
    dones_buffer = np.zeros((horizon, n), dtype=np.float32)

    if vec_env.last_obs is None:
        vec_env.last_obs = vec_env.reset()
    observations = vec_env.last_obs

    for t in range(horizon):
        obs_tensor = torch.as_tensor(observations, dtype=torch.float32)
        actions, log_probs, values = model.act(obs_tensor)

        obs_buffer[t] = observations
        actions_buffer[t] = actions.numpy()
        log_probs_buffer[t] = log_probs.numpy()
        values_buffer[t] = values.numpy()

        touched_before = vec_env.touched_flags()
        next_obs, rewards, terminated, truncated, final_obs = vec_env.step(
            actions_buffer[t]
        )
        scored = [team is not None for team in vec_env.scoring_teams()]

        rewards_buffer[t] = rewards
        dones_buffer[t] = (terminated | truncated).astype(np.float32)

        # The bootstrap value: 0 on TERMINATION, V(final_obs) on TRUNCATION,
        # V(next_obs) otherwise. This is the distinction that keeps value
        # learning honest.
        bootstrap = np.zeros(n, dtype=np.float32)
        ongoing = ~(terminated | truncated)
        if ongoing.any():
            bootstrap[ongoing] = model.predict_values(
                torch.as_tensor(next_obs[ongoing], dtype=torch.float32)
            ).numpy()
        for i in range(n):
            if truncated[i] and not terminated[i]:
                bootstrap[i] = float(model.predict_values(
                    torch.as_tensor(final_obs[i], dtype=torch.float32).unsqueeze(0)
                ).item())
        bootstraps_buffer[t] = bootstrap

        for i in range(n):
            tracker.record(float(rewards[i]), bool(terminated[i]), bool(truncated[i]),
                           touched_before[i], scored[i] and bool(terminated[i]))

        observations = next_obs

    vec_env.last_obs = observations
    return {
        "obs": obs_buffer, "actions": actions_buffer, "log_probs": log_probs_buffer,
        "values": values_buffer, "rewards": rewards_buffer,
        "bootstraps": bootstraps_buffer, "dones": dones_buffer,
    }

# train.py
"""Wires the MiniRL domain into core and trains it."""
import argparse
import copy
from typing import Dict, List

import numpy as np
import torch

from core.dashboard import Dashboard
from core.metrics import EpisodeTracker
from core.ppo import ActorCritic, PPOConfig, compute_gae, ppo_update
from core.rollout import VecEnv, collect_rollout
from domains.minirl.env import N_ACTIONS, OBS_SIZE, build_env
from domains.minirl.render import draw_arena


def _flatten(batch: Dict[str, np.ndarray], advantages: np.ndarray,
             returns: np.ndarray) -> Dict[str, torch.Tensor]:
    return {
        "obs": torch.as_tensor(batch["obs"].reshape(-1, OBS_SIZE), dtype=torch.float32),
        "actions": torch.as_tensor(batch["actions"].reshape(-1), dtype=torch.long),
        "old_log_probs": torch.as_tensor(batch["log_probs"].reshape(-1),
                                         dtype=torch.float32),
        "advantages": torch.as_tensor(advantages.reshape(-1), dtype=torch.float32),
        "returns": torch.as_tensor(returns.reshape(-1), dtype=torch.float32),
    }


def _record_episode(model: ActorCritic, max_steps: int = 450) -> List[object]:
    """Play one episode greedily and capture states for playback."""
    env = build_env()
    obs = env.reset()
    states = [copy.deepcopy(env.state)]
    for _ in range(max_steps):
        with torch.no_grad():
            logits = model.policy(torch.as_tensor(obs[0], dtype=torch.float32).unsqueeze(0))
            action = int(torch.argmax(logits, dim=-1).item())
        obs, _, terminated, truncated = env.step({0: action})
        states.append(copy.deepcopy(env.state))
        if terminated[0] or truncated[0]:
            break
    return states


def run_training(iterations: int = 400, n_envs: int = 8, horizon: int = 256,
                 render_every: int = 10, show_dashboard: bool = True,
                 seed: int = 0) -> List[Dict[str, float]]:
    torch.manual_seed(seed)
    np.random.seed(seed)

    config = PPOConfig()
    model = ActorCritic(OBS_SIZE, N_ACTIONS)
    optimizer = torch.optim.Adam(model.parameters(), lr=3e-4)
    vec_env = VecEnv(build_env, n_envs=n_envs)
    tracker = EpisodeTracker()
    dashboard = Dashboard(draw_fn=draw_arena) if show_dashboard else None

    history: List[Dict[str, float]] = []
    for iteration in range(1, iterations + 1):
        batch = collect_rollout(vec_env, model, horizon=horizon, tracker=tracker)
        advantages, returns = compute_gae(
            batch["rewards"], batch["values"], batch["bootstraps"], batch["dones"],
            gamma=config.gamma, lam=config.lam,
        )
        stats = ppo_update(model, optimizer, _flatten(batch, advantages, returns), config)
        snapshot = tracker.snapshot()
        history.append(snapshot)

        print(f"iter {iteration:4d} | reward {snapshot['mean_episode_reward']:8.3f} "
              f"| touch {snapshot['touch_rate']:5.2f} | goal {snapshot['goal_rate']:5.2f} "
              f"| entropy {stats['entropy']:5.3f} | episodes {snapshot['episodes']}")

        if dashboard is not None:
            dashboard.update_curve(iteration, snapshot)
            if render_every and iteration % render_every == 0:
                dashboard.show_episode(_record_episode(model))

    return history


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Train a MiniRL agent.")
    parser.add_argument("--iterations", type=int, default=400)
    parser.add_argument("--n-envs", type=int, default=8)
    parser.add_argument("--horizon", type=int, default=256)
    parser.add_argument("--render-every", type=int, default=10)
    parser.add_argument("--no-dashboard", action="store_true")
    parser.add_argument("--seed", type=int, default=0)
    args = parser.parse_args()

    run_training(iterations=args.iterations, n_envs=args.n_envs, horizon=args.horizon,
                 render_every=args.render_every, show_dashboard=not args.no_dashboard,
                 seed=args.seed)
    input("Training finished. Press Enter to close the dashboard.")
```
